# Runtime dependencies for the ingestion pipeline and chat backend.
#
# Install into the project venv (never system Python):
#     python -m venv .venv
#     .venv/Scripts/python -m pip install -r requirements.txt      # Windows
#     .venv/bin/python     -m pip install -r requirements.txt      # Linux/macOS
#
# Direct dependencies only, pinned to the versions actually resolved and
# tested against on Python 3.13. Transitive dependencies are left to the
# resolver; run `pip freeze > requirements.lock.txt` if byte-exact
# reproduction is ever needed.
#
# NOTE: infra/postgres-spm/apply_migrations.py deliberately uses NO
# third-party packages, so the database can be rebuilt before this venv
# exists — by anyone who has only Docker and a stock Python.

# ---- Database ---------------------------------------------------------
psycopg[binary]==3.3.4                  # Postgres driver (psycopg3)
psycopg_pool==3.3.1                     # connection pool for the chat
                                        # application only — the pipeline
                                        # opens one connection per run and
                                        # needs none. It is here because
                                        # pooling is precisely where the
                                        # SET LOCAL rule earns its keep: a
                                        # plain SET would persist onto the
                                        # next request to borrow that
                                        # connection, and that leak cannot
                                        # be reproduced — or tested —
                                        # without a real pool.

# ---- Vector store -----------------------------------------------------
qdrant-client==1.19.0

# ---- Task queue -------------------------------------------------------
celery[redis]==5.6.3                    # see SOLUTION_ARCHITECTURE.md 4.5

# ---- Azure model providers -------------------------------------------
azure-ai-documentintelligence==1.0.2    # OCR: prebuilt-layout
openai==2.53.0                          # Azure OpenAI: embeddings + LLM

# ---- Native Office fallback (pipeline/native_extract.py) --------------
# NOT the primary extraction path. Document Intelligence handles every
# format; these are used only when it REJECTS an Office file, which it
# does to valid .docx files repackaged by tools other than Word — the
# error even claims the content is corrupted when it demonstrably isn't.
# Without them such a document contributes nothing at all.
python-docx==1.2.0
python-pptx==1.0.2

# pdfplumber is deliberately NOT here. It was installed to measure native
# PDF extraction against Document Intelligence, and the measurement is why
# there is no PDF fallback: our own generated Arabic PDF yielded 36 Arabic
# characters against DI's 2,623, because text comes back as Arabic
# presentation forms in visual order, and an English PDF came back with
# every letter doubled. It may still be present in a developer venv from
# that experiment — that is not a dependency, and PDF text must not be
# extracted this way.

# ---- Chunking ---------------------------------------------------------
tiktoken==0.13.0                        # token counting. Swap for a HF
                                        # tokenizer when moving to an
                                        # on-prem embedding model —
                                        # SOLUTION_ARCHITECTURE.md 4.6

# ---- Language identification (AC01) -----------------------------------
py3langid==0.3.0                        # Latin-script language ID only.
                                        # Script detection (Arabic vs Han vs
                                        # Cyrillic) is done by counting
                                        # codepoints and needs nothing; this
                                        # exists solely because Latin script
                                        # cannot separate English from French
                                        # or Spanish. Offline, deterministic,
                                        # ~2MB embedded model, ~70ms on first
                                        # call and negligible after. Chosen
                                        # over lingua-py, which is far larger
                                        # for accuracy this branch does not
                                        # need. It is injectable at the one
                                        # call site in Application/backend/
                                        # language.py, so a swap stays there.

# ---- Config -----------------------------------------------------------
python-dotenv==1.2.2                    # loads infra/.env

# LangChain is deliberately NOT here yet. It is the chosen RAG
# orchestration layer (SOLUTION_ARCHITECTURE.md 4.5) but the ingestion
# pipeline doesn't need it — it belongs with the chat backend, and
# installing a large unused dependency tree now would only add churn.
